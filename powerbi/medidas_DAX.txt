    CALENDAR(DATE(2024,1,1), DATE(2027,12,31)),
    "Anio", YEAR([Date]), "Mes", MONTH([Date]),
    "Etiqueta", FORMAT([Date], "yyyy-mm"),
    "Tipo", IF([Date] > DATE(2026,9,30), "Proyeccion", IF([Date] >= DATE(2026,1,1), "2026 parcial", "Historico"))
)

dim_producto = DISTINCT(SELECTCOLUMNS(fact_precios, "producto", fact_precios[producto]))

dim_provincia = DISTINCT(SELECTCOLUMNS(fact_precios, "provincia", fact_precios[provincia]))

-- ============ MEDIDAS: DATOS ============
Registros = COUNTROWS(fact_precios)
Estaciones = DISTINCTCOUNT(fact_precios[idempresa])
Precio Mediano = MEDIAN(fact_precios[precio])
Precio Relativo Promedio = AVERAGE(fact_precios[precio_relativo])
Participacion Estaciones % =
    DIVIDE([Estaciones], CALCULATE([Estaciones], ALL(fact_precios[empresabandera])))

-- ============ MEDIDAS: REGRESION (filtrar predicciones[conjunto] = "Evaluacion 2026") ============
MAE =
    AVERAGEX(predicciones, ABS(predicciones[precio_predicho] - predicciones[precio]))
RMSE =
    SQRT(AVERAGEX(predicciones, (predicciones[precio_predicho] - predicciones[precio]) ^ 2))
R2 =
VAR media = AVERAGE(predicciones[precio])
RETURN 1 - DIVIDE(
    SUMX(predicciones, (predicciones[precio] - predicciones[precio_predicho]) ^ 2),
    SUMX(predicciones, (predicciones[precio] - media) ^ 2))
MAE (indice de mercado conocido) =
    AVERAGEX(predicciones, ABS(predicciones[precio_predicho_indice_conocido] - predicciones[precio]))

-- ============ MEDIDAS: CLASIFICACION (clase 1 = Cara) ============
VP = COUNTROWS(FILTER(predicciones, predicciones[clase] = 1 && predicciones[clase_predicha] = 1))
VN = COUNTROWS(FILTER(predicciones, predicciones[clase] = 0 && predicciones[clase_predicha] = 0))
FP = COUNTROWS(FILTER(predicciones, predicciones[clase] = 0 && predicciones[clase_predicha] = 1))
FN = COUNTROWS(FILTER(predicciones, predicciones[clase] = 1 && predicciones[clase_predicha] = 0))
Exactitud = DIVIDE([VP] + [VN], [VP] + [VN] + [FP] + [FN])
Precision Cara = DIVIDE([VP], [VP] + [FP])
Recall Cara = DIVIDE([VP], [VP] + [FN])
F1 Cara = DIVIDE(2 * [Precision Cara] * [Recall Cara], [Precision Cara] + [Recall Cara])

-- ============ MEDIDAS: PROYECCION ============
Precio Proyectado = AVERAGE(proyeccion_2027[precio_predicho])
Aviso = "Escenario estadistico. No es dato real ni precio oficial."
